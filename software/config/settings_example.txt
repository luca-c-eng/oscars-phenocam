# OSCARS-PHENOCAM dev/v0.1.0 settings example
# Acquisition timing and ICOS upload configuration.
#
# Positional format: one effective value per field.
# Blank lines and comment lines are ignored.
# Do not use a blank line to represent an empty field.
# Do not change the order of the values.
#
# The first six effective values are mandatory.

# 1 SITENAME
# Used in filenames, metadata and remote paths.
mysite

# 2 UTC_OFFSET
# Fixed station offset. Examples: 0, +1, -5, +5:30.
+1

# 3 TZ_LABEL
# Required by the positional parser, but regenerated from UTC_OFFSET.
UTC+1

# 4 START_HOUR
# Acquisition-window start, inclusive.
6

# 5 END_HOUR
# Acquisition-window end, exclusive.
22

# 6 INTERVAL_MIN
# Default acquisition interval in minutes.
# Supported values: 5, 10, 15, 20, 30 or 60.
30

# 7 IFACE
# auto or an explicit interface name.
auto

# 8 LEGACY_REMOTE_USER
# Reserved positional field retained for compatibility; not used by this
# ICOS-dedicated version.
phenocam

# 9 NET_MODE
# auto, ethernet or wifi.
auto

# 10 RAM_MIN_FREE_MB
# Minimum RAM-queue free space before spillover.
20

# 11 SD_MAX_USED_PCT
# Applied only when fallback to the SD queue is required.
80

# 12 USB_MOUNT_BASES
# Colon-separated mount locations scanned during regular capture and upload.
/media:/mnt

# 13 USB_MAX_USED_PCT
# USB is selected only when its usage is below this value.
90

# 14 REMOTE_LAYOUT
# Use icos for the Carbon Portal workflow.
icos

# 15 SITE_LAT
# Written as lat in metadata.
nd

# 16 SITE_LON
# Written as lon in metadata.
nd

# 17 SITE_ELEV_M
# Written as elev in metadata.
nd

# 18 SITE_START_DATE
# Written as start_date in metadata.
nd

# 19 SITE_END_DATE
# Written as end_date in metadata.
nd

# 20 SITE_NIMAGE
# Written as nimage in metadata.
nd

# 21 BOARD
# Read and exported, but not consumed by the runtime scripts.
unknown

# 22 CAMERA_MODEL
# Read and exported, but not consumed by the runtime scripts.
imx708

# 23 CAPTURE_TIMEOUT
# Camera warm-up time in milliseconds.
# A non-numeric value is replaced with 30000.
30000

# 24 VISION_EDGE_ENABLED
# Enables or disables Phenocam Vision Edge.
# Accepted values: on, off. Default: off.
off

# 25 VISION_EDGE_MODE
# Selects the action performed when detection is enabled.
#   metadata                - record detections without changing the JPEG
#   annotated               - draw detected objects on the JPEG
#   delete                  - discard the image when an object is detected
# Default: metadata.
metadata


# 26 IMAGE_WIDTH
# Image width in pixels.
# Supported pairs are based on the Camera Module 3 IMX708 sensor modes
# documented by Raspberry Pi:
#   1536x864  - uses a central 3072x1728 sensor area
#   2304x1296 - uses the full 4608x2592 sensor area; OSCARS default
#   4608x2592 - full sensor resolution
# IMAGE_WIDTH and IMAGE_HEIGHT must form one of these complete pairs.
2304

# 27 IMAGE_HEIGHT
# Image height in pixels.
# Must be paired with IMAGE_WIDTH as documented above.
1296

# 28 IMAGE_QUALITY
# JPEG quality accepted by rpicam-still: integer from 1 to 100.
# OSCARS initial default: 100.
# The recommended operational value will be established through tests.
100

# 29 CP_ENVIRONMENT
# Carbon Portal environment: staging or production.
# Keep staging until the complete workflow has passed acceptance tests.
staging

# 30 CP_SUBMITTER_ID
# Assigned by Carbon Portal technical staff. Upload remains disabled while nd.
nd

# 31 CP_STATION_URI
# Canonical ICOS station URI assigned to this installation.
nd

# 32 CP_INSTRUMENT_URI
# Canonical ICOS instrument URI assigned to this phenocamera.
nd

# 33 ICOS_FILENAME_SUFFIX
# Configurable final component of the daily ICOS filename.
# Use the station-specific value confirmed by ICOS, following the current
# Lxx_Fxx convention (for example L09_F01). Upload remains disabled while nd.
nd

# 34 CAPTURE_MINUTE
# Persistent offset within INTERVAL_MIN. For the default interval 30,
# 13 requests captures at minutes 13 and 43 of every hour.
# A new installation generates and stores one random value from 0 to 29.
13
