# OSCARS-PHENOCAM dev/v1.7.0 FTP credentials example

#

# Runtime destination:

# /etc/phenocam/ftp_credentials.txt

#

# Positional format: one effective value per field.

# Blank lines and comment lines are ignored.

# At least five effective values are required.

# Do not change their order.

#

# Replace every placeholder before using this file.

# The uploader rejects the known placeholder host, port, username and password.

#

# FTP credentials are stored as plain text.

# Never commit real credentials to the repository.

# 1 FTP_HOST

YOUR_FTP_HOST_OR_IP

# 2 FTP_PORT

# Must contain only digits.

# The uploader always uses ftp://; port 22 does not enable SFTP.

YOUR_FTP_PORT

# 3 FTP_REMOTE_BASE

/your/remote/base/path

# 4 FTP_USER

your_ftp_username

# 5 FTP_PASS

your_ftp_password

# Remote paths are generated from REMOTE_LAYOUT in settings.txt.

#

# general:

# FTP_REMOTE_BASE/SITENAME/YYYY/MM/<filename>

#

# icos:

# FTP_REMOTE_BASE/data/SITENAME/<filename>

#

# The uploader invokes curl with passive FTP and remote-directory creation.
